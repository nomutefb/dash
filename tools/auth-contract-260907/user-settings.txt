async function renderSheetList(slug){
  const cfg=SHEET_CONFIG[slug];
  try{
    const d=await api('GET','/api/sheet/'+slug);
    const rows=(d.rows||[]).filter(r=>cfg.columns.some(c=>r[c.key]!==''&&r[c.key]!=null));
    _sheetCache[slug]=rows;
    document.getElementById('sheet-admin-body').innerHTML=buildSheetListHtml(slug,rows);
  }catch(e){
    document.getElementById('sheet-admin-body').innerHTML='<div style="color:red;padding:20px">로드 실패: '+escapeHtml(e.message)+'</div>';
  }
}

  }finally{
    if(typeof hideOverlayLoad==='function')hideOverlayLoad();

// [260906 입력절차③-2] 시트 저장은 열 이름으로 — 위치 배열은 시트 열 순서(담당자 18열)와 폼 열(16열)이 어긋나면 값이 밀린다(회계여부 → 전화번호 자리).
function _sheetValuesObj(slug,values){ var o={}; ((SHEET_CONFIG[slug]||{}).columns||[]).forEach(function(c,i){ o[c.key]=values[i]; }); return o; }
async function saveSheetRow(slug,rowIndex){
  const form=document.getElementById('sheet-form');
  if(!form)return;
  const cfg=SHEET_CONFIG[slug];
  const fd=new FormData(form);
  const _existRow=rowIndex?(_sheetCache[slug]||[]).find(r=>r._rowIndex===rowIndex):null;
  // [260613] 관리자 해제 시 비밀번호도 제거 — 관리자였다가 내려간 계정에 비번이 남던 버그
  const _mgrAdminEl=(slug==='manager')?form.elements['관리자여부']:null;
  const _mgrIsAdmin=_mgrAdminEl?_mgrAdminEl.checked:(_existRow?isFlagOn(_existRow['관리자여부']):false);
  const values=cfg.columns.map(c=>{
    const el=form.elements[c.key];
    if(c.type==='checkbox'){
      // 폼에 필드가 없으면(등록폼에서 제외된 컬럼): 활성=신규는 '1' 강제, 그 외는 기존값/빈값 보존
      if(!el){
        if(slug==='manager'&&c.key==='계정여부')return _existRow?(String(_existRow[c.key]||'').trim()||'1'):'1';
        return _existRow?(_existRow[c.key]||''):'';
      }
      return el.checked?'1':'';
    }
    // 텍스트: 폼에 없으면(PIN/비밀번호 등 제외 컬럼) 기존값 보존, 신규면 빈값
    if(!el){
      if(slug==='manager'&&c.key==='비밀번호'&&!_mgrIsAdmin)return '';  // 관리자 아니면 비밀번호 비움
      return _existRow?(_existRow[c.key]||''):'';
    }
    return fd.get(c.key)||'';
  });
  showOverlayLoad('\uC800\uC7A5 \uC911\uC785\uB2C8\uB2E4...');
  try{
    if(rowIndex)await api('PATCH','/api/sheet/'+slug+'/'+rowIndex,{values:_sheetValuesObj(slug,values)});
    else await api('POST','/api/sheet/'+slug,{values:_sheetValuesObj(slug,values)});
    // 콘텐츠/담당자/플랫폼 변경 시 작성 모달 마스터 캐시 무효화 & 재로드
    if(slug==='content'){CONTENT_TYPES=[];await loadContentMeta();populateMasterDropdowns();}
    if(slug==='manager'){MANAGERS=[];await loadManagers();PR_MANAGERS=MANAGERS.filter(m=>isFlagOn(m['홍보여부']));populateMasterDropdowns();}
    if(slug==='platform'){await loadPlatforms();populateMasterDropdowns();}
    await renderSheetList(slug);
    closeSheetForm();
    showToast('\uC800\uC7A5\uB418\uC5C8\uC2B5\uB2C8\uB2E4','success');
  }catch(e){
    showToast('\uC800\uC7A5 \uC2E4\uD328: '+e.message,'error');
  }finally{
    hideOverlayLoad();
  }
}
