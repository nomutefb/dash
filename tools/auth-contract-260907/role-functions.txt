async function bulkApprovePending(){
  if(!_pendingChecks.size){showToast('선택된 건이 없어요','error');return;}
  const ids = Array.from(_pendingChecks);
  const ok = await showConfirm('선택한 '+ids.length+'건을 접수하시겠어요?<br><br><span style="color:#888;font-size:12px">신청 중 → 예정으로 변경됩니다.</span>',{title:'일괄 접수',html:true});
  if(!ok) return;
  showOverlayLoad('일괄 접수 중 (0/'+ids.length+')');
  let success = 0, failed = 0, skipped = 0, _failReason = '';
  for(let i = 0; i < ids.length; i++){
    const rowIndex = ids[i];
    try{
      const rec = records.find(function(r){return r._rowIndex === rowIndex;});
      if(!rec){failed++;continue;}
      // [버그수정] 단건 접수와 동일 가드 — 신청 중 상태만 + 카카오/문자 충돌 검증 (일괄 경로 우회 차단)
      const _st=(rec['진행 상태']||'').toString().trim();
      if(_st!=='신청 중'){skipped++;continue;}
      const _ca=canApprove(rec,rowIndex);
      if(!_ca.ok){failed++;if(!_failReason)_failReason=_ca.reason;continue;}
      const row = buildRecordRow(rec, {'진행 상태': '예정', '재신청사유': ''});
      await api('PATCH', '/api/records/'+rowIndex, {values: row});
      success++;
      // 메시지함 hook — 단건 접수와 동일하게 신청자에게 알림
      try{
        pushMessage({recipient:rec['신청자']||'',type:'일반',trigger:'신청중→예정',before:'신청 중',after:'예정',reason:'',refNo:rec['No']||rec['NO']||'',refSummary:(getRecDateKey(rec)+' '+(rec['플랫폼 1']||'')+' '+(rec['콘텐츠 제목']||'')).trim()});
      }catch(_e){ console.error('[pushMessage bulkApprove]', _e); }
      showOverlayLoad('일괄 접수 중 ('+(i+1)+'/'+ids.length+')');
    }catch(e){ console.error(e); failed++; }
  }
  hideOverlayLoad();
  showToast(success+'건 접수 완료'+(skipped?' / '+skipped+'건 대상 아님':'')+(failed?' / '+failed+'건 실패':'')+(_failReason?' — '+_failReason:''), failed?'error':'success');
  _pendingChecks.clear();
  await syncReload(0);
  if(typeof renderPromoBoard==='function') renderPromoBoard();
  render();
}

  const todayActiveCnt=records.filter(function(r){return getRecDateKey(r)===todayKey&&r['진행 상태']!=='완료'&&r['진행 상태']!=='취소'&&r['진행 상태']!=='임시';}).length;

  if(rows.length===0){

  if(!editing?.existing)return;
  if(userRole!=='admin'){showToast('관리자만 할 수 있어요','error');return;}
  const ok=await showConfirm('이 홍보 신청을 승인할까요?');if(!ok)return;
  showOverlayLoad();
  try{
    // 진행 상태 = "예정"으로 변경 + 시간/날짜 등은 modal 폼에서 현재 값으로
    const _apRec=editing.existing;
    // 현재 모달 폼의 값으로 갱신 (관리자가 시간/날짜 조정 가능)
    document.getElementById('f-status').value='예정';
    const _saved=await saveEntry();  // 기존 row in-place PATCH
    if(_saved){
      // 메시지함 hook — 접수(승인) 알림 (다른 승인 경로와 동일)
      try{
        pushMessage({recipient:_apRec['신청자']||'',type:'일반',trigger:'신청중→예정',before:_apRec['진행 상태']||'신청 중',after:'예정',reason:'',refNo:_apRec['No']||_apRec['NO']||'',refSummary:(getRecDateKey(_apRec)+' '+(_apRec['플랫폼 1']||'')+' '+(_apRec['콘텐츠 제목']||'')).trim()});
      }catch(_e){ console.error('[pushMessage approvePromo]', _e); }
      showToast('승인됐어요');
    }
  }catch(e){showToast('승인 실패: '+e.message,'error');}
  finally{hideOverlayLoad();}
}

async function rejectPromo(){
  if(!editing?.existing)return;
  if(userRole!=='admin'){showToast('관리자만 할 수 있어요','error');return;}
  // 사유 입력 (반려는 record 삭제 → 신청자 메시지에 사유 전달 필수)
  const reason=await _renderCancelReasonModal(1,{
    title:'신청 반려',
    subtitle:'반려 사유를 입력해 주세요. 신청자에게 전달되며 기록이 삭제됩니다.',
    labelText:'반려 사유',
    placeholder:'예: 동일 일정 중복 / 정책 부적합 / 자료 불충분 등',
    buttonLabel:'반려 (삭제)',
    buttonColor:'var(--danger-btn)'
  });
  if(!reason)return;
  // DELETE 전 record 정보 캐시 (메시지 푸시용)
  const _rec = editing.existing;
  const _refNo = _rec['No']||_rec['NO']||'';
  const _refApplicant = _rec['신청자']||'';
  const _refSum = (_rec['날짜']||'') + ' ' + (_rec['플랫폼 1']||'') + ' ' + (_rec['콘텐츠 제목']||'');
  showOverlayLoad();
  try{
    const ri=_rec._rowIndex;
    await api('POST','/api/records-delete',{rowIndex:ri});
    // 메시지함 hook (중요: 신청→반려, 신청자에게 사유 전달)
    try{
      if(_refApplicant){
        pushMessage({
          recipient: _refApplicant,
          type: '중요',
          trigger: '신청→반려',
          before: _rec['진행 상태']||'신청 중',
          after: '반려 (삭제)',
          reason: reason,
          refNo: _refNo,
          refSummary: _refSum.trim()
        });
      }
    }catch(_e){ console.error('[pushMessage reject]', _e); }
    closeModal();
    await syncReload(-1);
    closePanel();
    showToast('반려 처리됐어요 (사유 전달됨)');
  }catch(e){showToast('반려 실패: '+e.message,'error');}
  finally{hideOverlayLoad();}
}

async function submitPromoRequest(){
  // ===== 신청 직전 재검증 (통합 게이트: 접수/홍보시간대/제외/담당자/판매기간/카카오문자) =====
  const _v=_validateScheduleChange(null,_pwData.date,_pwData.time,{action:'submit',plat1:_pwData.plat1,plat2:_pwData.plat2,applicant:_pwData.applicant,program:_pwData.program});
  if(!_v.ok){showToast('신청 불가: '+_v.reason,'error');return;}
  showOverlayLoad();
  try{
    // 임시저장 이어쓰기 제출이면 그 행을 PATCH해 전환 (No 보존 · 중복 행 방지)
    const _draftRec=_pwData._draftRow?records.find(function(r){return r._rowIndex===_pwData._draftRow;}):null;
    var _batchOff=(_pwBatchQueue!=null)?Math.max(0,_pwBatchTotal-_pwBatchQueue.length-1):0;  // 배치 순차: syncReload 전이라 records 미갱신 → No 오프셋(중복 방지)
    const _subNo=_draftRec?(_draftRec['No']||_draftRec['NO']||records.length+1):(records.length+1+_batchOff);
    // 29컬럼 row = 공용 빌더(단건/임시저장/배치 단일 진실원). 진행상태 admin='예정'/user='신청 중'
    const row=_buildRecRow(_pwData, (userRole==='admin'?'예정':'신청 중'), _subNo, false);
    if(_pwData._draftRow){
      await api('PATCH','/api/records/'+_pwData._draftRow,{values:row});
    }else{
      await api('POST','/api/records',{values:row});
    }
    // ⚠️ myApplicant는 절대 _pwData.applicant(게시 담당자, P열)로 덮어쓰지 않음.
    // 로그인 시 line 1400에서 본인 이름으로 이미 저장됨. 위저드에서 갱신 불필요.
    // 이전 코드 (if userRole!=='admin' setItem _pwData.applicant)는
    // user의 myApplicant를 게시 담당자 값으로 오염시키는 버그였음.
    // 예: Cameron Lane 09790이 게시 담당자=Avery Reed 08336으로 신청 → localStorage가 'Avery Reed 08336'으로 덮어씌워짐
    // → UI가 "Avery Reed 08336 님"으로 바뀌고 "내 신청만" 필터 작동 안 함.
    if(_pwBatchQueue!=null){ hideOverlayLoad(); _pwRunNextJob(); return; }  // [배치 순차] syncReload·완료화면은 큐 끝에서 1회
    await syncReload();
    _pwStep=99;
    renderPwStep();
    setTimeout(function(){
      const acts=document.getElementById('pw-acts');
      if(acts)acts.innerHTML='<button class="m-btn save" onclick="closePromoWizard(true);openPromoBoard()" style="margin-left:auto">신청 내역 보기</button>';
    },50);
  }catch(e){
    showToast('신청 실패: '+e.message,'error');
  }finally{
    hideOverlayLoad();
  }
}
