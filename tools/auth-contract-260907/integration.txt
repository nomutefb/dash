// 시트 적재 — 시트가 아직 없으면 `/api/ops`가 `note:'시트 없음 (미동기화)'` + 빈 rows로 답한다(에러 아님)
//   = 첫 사용자는 **씨앗만으로** 정상 렌더된다. 시트 신설은 첫 저장(`opsWriteSheet`→`ensureSheet`)이 한다.
function _finLoad(force){
  if(_finState.rows&&!force)return Promise.resolve(_finState.rows);
  if(_finState.busy&&_finState._p&&!force)return _finState._p;
  _finState.busy=1;
  _finState._p=api('GET','/api/ops?sheet='+encodeURIComponent(_FIN_SHEET)+(force?'&fresh=1':''))
    .then(function(d){ _finState.rows=(d&&d.rows)?d.rows:[]; _finState.note=(d&&d.note)||null; _finState.err=null; })
    .catch(function(e){ _finState.rows=[]; _finState.note=null; _finState.err=e; console.warn('[사업비]',e); })
    .then(function(){ _finState.busy=0; _finIdxClear(); return _finState.rows; });
  return _finState._p;
}
