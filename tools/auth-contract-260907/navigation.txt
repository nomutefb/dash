function _updateSalesAsOfBadge(data){
  var el=document.getElementById('sales-asof-badge');
  if(!el)return;
  try{
    var rows=data&&Array.isArray(data.rows)?data.rows:[], max='';
    rows.forEach(function(row){
      var raw=String(row&&row['기준일자']||'').replace(/[^0-9]/g,'');
      if(/^\d{8}$/.test(raw)&&raw>max)max=raw;
    });
    if(!max){el.textContent='';el.style.display='none';return;}
    el.textContent='판매 데이터 '+parseInt(max.slice(4,6),10)+'/'+parseInt(max.slice(6,8),10)+' 기준';
    el.style.display='inline-block';
  }catch(_e){el.textContent='';el.style.display='none';}
}

// 빌더의 구 인라인 onmouseenter/leave 배경 스왑은 pill이 대체(속성 핸들러를 프로퍼티로 덮음) — '기본' 틴트 항목은 자기 배경 유지 + pill 겹침(구 호버 진하기와 동일 체감).
function _ddPanel(builder,minW){
  const p=document.createElement('div');p.className='dd-panel';p.style.minWidth=(minW||176)+'px';
  builder(p);
  const pill=document.createElement('div');pill.className='dd-pill';p.insertBefore(pill,p.firstChild);
  p.querySelectorAll('button').forEach(function(b){
    b.onmouseenter=function(){const vis=pill.classList.contains('is-visible');
      if(!vis)pill.classList.add('is-instant');   // 첫 등장 = 제자리 페이드(레퍼런스 is-instant 문법)
      pill.style.top=b.offsetTop+'px';pill.style.left=b.offsetLeft+'px';pill.style.width=b.offsetWidth+'px';pill.style.height=b.offsetHeight+'px';
      if(!vis){void pill.offsetWidth;pill.classList.add('is-visible');requestAnimationFrame(function(){pill.classList.remove('is-instant');});}
    };
    b.onmouseleave=null;
  });
  p.onmouseleave=function(){pill.classList.remove('is-visible');};
  return p;
}

//   우측 끝 대메뉴에서 폭이 넘칠 땐 화면 안으로 당김(양쪽 여백 12px) — 잘림 방지.
function _ddLeftFor(r,w){ return Math.max(12, Math.min(r.left, window.innerWidth-w-12)); }
// 메뉴 간 이동 = 닫고 새로 열지 않고 한 컨테이너가 위치·크기 모핑 + 패널 방향 슬라이드(레퍼런스 핵심 문법 — 이동 방향으로 이전 패널 퇴장·새 패널 입장)
function _ddMorph(m,anchor,builder,minW,owner){
  const menu=m._menu,old=menu.querySelector('.dd-panel.is-active');
  const dir=anchor.getBoundingClientRect().left>=m._anchor.getBoundingClientRect().left?1:-1;   // 1 = 오른쪽 메뉴로 이동
  const np=_ddPanel(builder,minW);
  np.classList.add(dir===1?'is-enter-right':'is-enter-left');
  menu.appendChild(np);
  const r=anchor.getBoundingClientRect();
  menu.classList.add('dd-morphing');
  menu.style.left=_ddLeftFor(r,np.offsetWidth)+'px';   // [260731 운영자] 메뉴 바로 아래 = 앵커 좌측 정렬(구 우측정렬 right)
  menu.style.width=np.offsetWidth+'px';menu.style.height=np.offsetHeight+'px';
  void np.offsetWidth;   // enter 초기 위치 커밋 후 전이 시작
  np.classList.remove('is-enter-right','is-enter-left');np.classList.add('is-active');
  if(old){old.classList.remove('is-active');old.classList.add(dir===1?'is-leaving-left':'is-leaving-right');setTimeout(function(){old.remove();},500);}
  document.querySelectorAll('.nav-btn[aria-expanded]').forEach(function(b){if(b!==anchor)b.removeAttribute('aria-expanded');});
  anchor.setAttribute('aria-expanded','true');
  m.dataset.owner=owner;m._anchor=anchor;
}
